This guide shows how to transfer players between servers, with proper error handling and user feedback.

## Basic Transfer

<CodeGroup>
```java Java
public void transferPlayer(UUID playerUuid, String targetServer) {
    api.player().get(playerUuid).thenAccept(player -> {
        if (player == null) {
            System.out.println("Player not found or offline");
            return;
        }

        player.connect(targetServer).thenAccept(result -> {
            switch (result) {
                case SUCCESS -> System.out.println("Transfer successful");
                case SERVER_NOT_FOUND -> System.out.println("Server doesn't exist");
                case ALREADY_CONNECTED -> System.out.println("Already on that server");
                case PLAYER_NOT_FOUND -> System.out.println("Player went offline");
                case CONNECTION_FAILED -> System.out.println("Transfer failed");
            }
        });
    });
}
```

```kotlin Kotlin
suspend fun transferPlayer(playerUuid: UUID, targetServer: String) {
    val player = api.player().get(playerUuid).await()
    if (player == null) {
        println("Player not found or offline")
        return
    }

    val result = player.connect(targetServer).await()
    when (result) {
        CloudPlayer.ConnectResult.SUCCESS -> println("Transfer successful")
        CloudPlayer.ConnectResult.SERVER_NOT_FOUND -> println("Server doesn't exist")
        CloudPlayer.ConnectResult.ALREADY_CONNECTED -> println("Already on that server")
        CloudPlayer.ConnectResult.PLAYER_NOT_FOUND -> println("Player went offline")
        CloudPlayer.ConnectResult.CONNECTION_FAILED -> println("Transfer failed")
    }
}
```
</CodeGroup>

## Transfer to Best Server in Group

Find the server with the most available slots:

<CodeGroup>
```java Java
public void transferToGroup(UUID playerUuid, String groupName) {
    api.server().getServersByGroup(groupName).thenAccept(servers -> {
        Server bestServer = servers.stream()
            .filter(s -> s.getState() == ServerState.AVAILABLE
                      || s.getState() == ServerState.INGAME)
            .filter(s -> s.getPlayerCount() != null
                      && s.getPlayerCount() < s.getMaxPlayers())
            .max(Comparator.comparingInt(s -> s.getMaxPlayers() - s.getPlayerCount()))
            .orElse(null);

        if (bestServer == null) {
            System.out.println("No available servers in group: " + groupName);
            return;
        }

        api.player().get(playerUuid).thenAccept(player -> {
            if (player != null) {
                player.connect(bestServer.getServerId());
            }
        });
    });
}
```

```kotlin Kotlin
suspend fun transferToGroup(playerUuid: UUID, groupName: String) {
    val servers = api.server().getServersByGroup(groupName).await()
    val bestServer = servers
        .filter { s -> s.state == ServerState.AVAILABLE || s.state == ServerState.INGAME }
        .filter { s -> s.playerCount != null && s.playerCount!! < s.maxPlayers }
        .maxByOrNull { s -> s.maxPlayers - (s.playerCount ?: 0) }

    if (bestServer == null) {
        println("No available servers in group: $groupName")
        return
    }

    val player = api.player().get(playerUuid).await() ?: return
    player.connect(bestServer.serverId).await()
}
```
</CodeGroup>

## Transfer with User Feedback

Send messages to the player during transfer:

<CodeGroup>
```java Java
public void transferWithFeedback(UUID playerUuid, String targetServer) {
    api.player().get(playerUuid).thenCompose(player -> {
        if (player == null) {
            return CompletableFuture.completedFuture(null);
        }

        player.sendMessage(Component.text("Connecting to " + targetServer + "...")
            .color(NamedTextColor.YELLOW));

        return player.connect(targetServer).thenApply(result -> {
            if (result != CloudPlayer.ConnectResult.SUCCESS) {
                player.sendMessage(Component.text("Failed to connect: " + result)
                    .color(NamedTextColor.RED));
            }
            return result;
        });
    });
}
```

```kotlin Kotlin
suspend fun transferWithFeedback(playerUuid: UUID, targetServer: String) {
    val player = api.player().get(playerUuid).await() ?: return

    player.sendMessage(Component.text("Connecting to $targetServer...")
        .color(NamedTextColor.YELLOW))

    val result = player.connect(targetServer).await()
    if (result != CloudPlayer.ConnectResult.SUCCESS) {
        player.sendMessage(Component.text("Failed to connect: $result")
            .color(NamedTextColor.RED))
    }
}
```

## Transfer All Players from Server

Evacuate all players before server shutdown:

<CodeGroup>
```java Java
public void evacuateServer(String serverId, String targetServer) {
    api.player().getOnlinePlayers().thenAccept(players -> {
        players.stream()
            .filter(p -> serverId.equals(p.getConnectedServerName()))
            .forEach(player -> {
                player.sendMessage(Component.text("Server shutting down, transferring...")
                    .color(NamedTextColor.GOLD));
                player.connect(targetServer);
            });
    });
}
```

```kotlin Kotlin
suspend fun evacuateServer(serverId: String, targetServer: String) {
    val players = api.player().getOnlinePlayers().await()
    players
        .filter { p -> serverId == p.connectedServerName }
        .forEach { player ->
            player.sendMessage(Component.text("Server shutting down, transferring...")
                .color(NamedTextColor.GOLD))
            player.connect(targetServer).await()
        }
}
```

## Request Server and Transfer Player

Request a new server to start, then transfer the player once it is ready:

<CodeGroup>
```java Java
public void requestServerAndTransfer(UUID playerUuid, String groupName) {
    api.server().getServersByGroup(groupName).thenAccept(servers -> {
        Server available = servers.stream()
            .filter(s -> s.getState() == ServerState.AVAILABLE)
            .filter(s -> s.getPlayerCount() != null
                      && s.getPlayerCount() < s.getMaxPlayers())
            .findFirst()
            .orElse(null);

        if (available != null) {
            api.player().get(playerUuid).thenAccept(player -> {
                if (player != null) player.connect(available.getServerId());
            });
        } else {
            // No available server; queue a start, then wait for it to become ready
            api.group().getGroupByName(groupName).thenAccept(group -> {
                if (group == null) return;

                api.group().requestServerStart(group).thenRun(() -> {
                    // Listen for the new server to become available
                    Subscription[] subHolder = new Subscription[1];
                    subHolder[0] = api.event().server().onStateChanged(event -> {
                        Server s = event.getServer();
                        if (s.isFromGroup()
                                && groupName.equals(s.getServerBase().getName())
                                && event.getNewState() == ServerState.AVAILABLE) {
                            subHolder[0].close();
                            api.player().get(playerUuid).thenAccept(player -> {
                                if (player != null) player.connect(event.getServerId());
                            });
                        }
                    });
                });
            });
        }
    });
}
```

```kotlin Kotlin
suspend fun requestServerAndTransfer(playerUuid: UUID, groupName: String) {
    val servers = api.server().getServersByGroup(groupName).await()
    val available = servers
        .filter { s -> s.state == ServerState.AVAILABLE }
        .filter { s -> s.playerCount != null && s.playerCount!! < s.maxPlayers }
        .firstOrNull()

    if (available != null) {
        val player = api.player().get(playerUuid).await() ?: return
        player.connect(available.serverId).await()
    } else {
        // No available server; queue a start, then wait for it to become ready
        val group = api.group().getGroupByName(groupName).await() ?: return
        api.group().requestServerStart(group).await()

        // Listen for the new server to become available
        val future = CompletableFuture<String>()
        val sub = api.event().server().onStateChanged { event ->
            val s = event.server
            if (s.isFromGroup()
                    && groupName == s.serverBase.name
                    && event.newState == ServerState.AVAILABLE) {
                future.complete(event.serverId)
            }
        }
        val newServerId = future.await()
        sub.close()

        val player = api.player().get(playerUuid).await() ?: return
        player.connect(newServerId).await()
    }
}
```
</CodeGroup>
