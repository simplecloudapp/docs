## Einfache Event-Behandlung

### Events abonnieren

<CodeGroup>
```java Java
// Subscribe gibt ein Subscription-Objekt zurück
Subscription sub = api.event().server().onStarted(event -> {
    System.out.println("Server gestartet: " + event.getServerId());
});

// Schließen wenn nicht mehr benötigt
sub.close();
```

```kotlin Kotlin

val sub = api.event().server().onStarted { event ->
```

### Plugin-Lifecycle-Verwaltung

<CodeGroup>
```java Java
public class MyPlugin extends JavaPlugin {
    private CloudApi api;
    private final List<Subscription> subscriptions = new ArrayList<>();

    @Override
    public void onEnable() {
        api = CloudApi.create();
        registerEvents();
    }

    @Override
    public void onDisable() {
        subscriptions.forEach(Subscription::close);
        subscriptions.clear();
    }

    private void registerEvents() {
        subscriptions.add(api.event().server().onStarted(this::onServerStarted));
        subscriptions.add(api.event().server().onStopped(this::onServerStopped));
        subscriptions.add(api.event().group().onUpdated(this::onGroupUpdated));
    }

    private void onServerStarted(ServerStartedEvent event) {
        getLogger().info("Server gestartet: " + event.getServerId());
    }

    private void onServerStopped(ServerStoppedEvent event) {
        getLogger().info("Server gestoppt: " + event.getServerId());
    }

    private void onGroupUpdated(GroupUpdatedEvent event) {
        getLogger().info("Group aktualisiert: " + event.getServerGroupId());
    }
}
```

## Gängige Event-Muster

### Reagieren, wenn eine Group keine laufenden Server mehr hat

<CodeGroup>
```java Java
public void setupAutoStart(String groupName, int minimum) {
    api.event().server().onStopped(event -> {
        Server server = event.getServer();
        if (!server.isFromGroup()) return;
        if (!groupName.equals(server.getServerBase().getName())) return;

        api.server().getServersByGroup(groupName).thenAccept(servers -> {
            long running = servers.stream()
                .filter(s -> s.getState() != ServerState.STOPPING
                          && s.getState() != ServerState.CLEANUP)
                .count();

            if (running < minimum) {
                api.group().getGroupByName(groupName).thenAccept(group -> {
                    if (group != null) {
                        api.group().requestServerStart(group);
                    }
                });
            }
        });
    });
}
```

```kotlin Kotlin
fun setupAutoStart(groupName: String, minimum: Int) {
    api.event().server().onStopped { event ->
        val server = event.server
        if (!server.isFromGroup()) return@onStopped
        if (groupName != server.serverBase.name) return@onStopped

        launch {
            val servers = api.server().getServersByGroup(groupName).await()
            val running = servers.count { s ->
                s.state != ServerState.STOPPING && s.state != ServerState.CLEANUP
            }

            if (running < minimum) {
                val group = api.group().getGroupByName(groupName).await() ?: return@launch
                api.group().requestServerStart(group).await()
            }
        }
    }
}
```

### Staff bei Server-Absturz benachrichtigen

<CodeGroup>
```java Java
public void setupCrashNotifications() {
    api.event().server().onStopped(event -> {
        if (!event.getCrashed()) return;

        String msg = "[Alert] Server abgestürzt: " + event.getServerId();
        if (event.getReason() != null) {
            msg += " — " + event.getReason();
        }
        notifyStaff(Component.text(msg).color(NamedTextColor.RED));
    });
}

private void notifyStaff(Component message) {
    api.player().getOnlinePlayers().thenAccept(players ->
        players.stream()
            .filter(p -> hasStaffPermission(p))
            .forEach(p -> p.sendMessage(message)));
}
```


## Einmalige Events

### Auf einen bestimmten Server warten, bis er verfügbar ist

<CodeGroup>
```java Java
public CompletableFuture<Void> waitForServer(String serverId) {
    CompletableFuture<Void> future = new CompletableFuture<>();

    Subscription sub = api.event().server().onStateChanged(event -> {
        if (event.getServerId().equals(serverId) &&
            event.getNewState() == ServerState.AVAILABLE) {
            future.complete(null);
        }
    });

    future.whenComplete((result, error) -> sub.close());

    return future;
}

// Verwendung
waitForServer("lobby-uuid").thenRun(() ->
    System.out.println("Server ist bereit!"));
```

```kotlin Kotlin
suspend fun waitForServer(serverId: String) {
    val future = CompletableFuture<Unit>()

    val sub = api.event().server().onStateChanged { event ->
        if (event.serverId == serverId &&
            event.newState == ServerState.AVAILABLE) {
            future.complete(Unit)
        }
    }

    future.await()
    sub.close()
}

// Verwendung
waitForServer("lobby-uuid")
println("Server ist bereit!")
```

### Mit Timeout

<CodeGroup>
```java Java
public CompletableFuture<Void> waitForServerWithTimeout(
        String serverId, Duration timeout) {
    CompletableFuture<Void> future = new CompletableFuture<>();

    Subscription sub = api.event().server().onStateChanged(event -> {
        if (event.getServerId().equals(serverId) &&
            event.getNewState() == ServerState.AVAILABLE) {
            future.complete(null);
        }
    });

    CompletableFuture.delayedExecutor(timeout.toMillis(), TimeUnit.MILLISECONDS)
        .execute(() -> future.completeExceptionally(
            new TimeoutException("Server wurde nicht rechtzeitig verfügbar")));

    future.whenComplete((result, error) -> sub.close());

    return future;
}
```
